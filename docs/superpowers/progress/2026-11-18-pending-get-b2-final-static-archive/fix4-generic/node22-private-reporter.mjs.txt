import { createHash } from 'node:crypto';
import { channel } from 'node:diagnostics_channel';
import { writeSync } from 'node:fs';
import { resolve } from 'node:path';

// Private pipe only. The parent consumes bounded JSON records before any public receipt.
const limits = { totalBytes: 1048576, messageBytes: 65536, events: 4096, children: 8 };
const monotonic = () => process.hrtime.bigint().toString();
let bytes = 0;
let sequence = 0;
let constructed = 0;
let inputBytes = 0;
const leftovers = new Map();
const digest = value => createHash('sha256').update(value, 'utf8').digest('hex');
const lifecycle = channel('child_process');

function emit(kind, fields) {
  const record = Buffer.from(JSON.stringify({ sequence: ++sequence, kind, ...fields }) + '\n');
  bytes += record.length;
  if (bytes > limits.totalBytes || sequence > limits.events) throw Error('OBSERVER_CAP');
  // For a pipe this may be short or EAGAIN. Incomplete records are unqualified.
  if (writeSync(1, record) !== record.length) throw Error('OBSERVER_SHORT_WRITE');
}

function location(file) {
  return typeof file === 'string' && file.length > 0 ? resolve(file) : null;
}

function watchChild({ process: child }) {
  const at = monotonic();
  const index = ++constructed;
  if (index > limits.children) throw Error('OBSERVER_CHILD_CAP');
  emit('construct', { index, at });
  let spawned = false;
  child.once('spawn', () => {
    spawned = true;
    emit('spawn', { index, at: monotonic(), pid: child.pid, file: location(child.spawnargs.at(-1)) });
  });
  child.once('exit', (code, signal) => {
    emit('exit', { index, at: monotonic(), pid: child.pid, spawned, code, signal });
  });
  child.once('close', (code, signal) => {
    emit('close', { index, at: monotonic(), pid: child.pid ?? null, spawned, code, signal });
  });
}

// Node 22.23.4 awaits custom reporter import in setup() before runFiles().
if (process.version !== 'v22.23.4' || process.env.NODE_TEST_CONTEXT !== undefined)
  throw Error('OBSERVER_REQUIRES_PINNED_OUTER_CLI');
lifecycle.subscribe(watchChild);
emit('ready', { at: monotonic(), pid: process.pid });

function readChunk(file, origin, chunk) {
  if (!file || typeof chunk !== 'string') throw Error('OBSERVER_INVALID_STREAM');
  inputBytes += Buffer.byteLength(chunk);
  if (inputBytes > limits.totalBytes) throw Error('OBSERVER_INPUT_CAP');
  const key = JSON.stringify([file, origin]);
  const parts = ((leftovers.get(key) ?? '') + chunk).split('\n');
  const carry = parts.pop();
  if (Buffer.byteLength(carry) > limits.messageBytes) throw Error('OBSERVER_LINE_CAP');
  if (carry) leftovers.set(key, carry);
  else leftovers.delete(key);
  for (const line of parts) {
    const value = line.endsWith('\r') ? line.slice(0, -1) : line;
    const length = Buffer.byteLength(value);
    if (length > limits.messageBytes) throw Error('OBSERVER_LINE_CAP');
    emit('line', { file, origin, length, sha256: digest(value) });
  }
}

function errorFields(error) {
  if (error == null) return null;
  const text = value => typeof value === 'string' ? value : null;
  const cause = error.cause;
  return {
    name: text(error.name), code: text(error.code), failureType: text(error.failureType),
    messageDigest: typeof error.message === 'string' ? digest(error.message) : null, causeName: text(cause?.name), causeCode: text(cause?.code),
    causeMessageDigest: typeof cause?.message === 'string' ? digest(cause.message) : null, exitCode: error.exitCode ?? null, signal: error.signal ?? null,
  };
}

export default async function* reporter(events) {
  try {
    for await (const { type, data } of events) {
      const file = location(data.file);
      if (type === 'test:stdout' || type === 'test:stderr') {
        readChunk(file, type, data.message);
      } else if (type === 'test:pass' || type === 'test:fail') {
        emit('test', { file, event: type, name: data.name, nesting: data.nesting,
          skipped: data.skip !== undefined, todo: data.todo !== undefined, error: errorFields(data.details?.error) });
      } else if (type === 'test:summary') {
        emit('summary', { file, success: data.success, counts: data.counts });
      } else if (type === 'test:diagnostic') {
        if (data.level === 'error' || typeof data.message !== 'string') throw Error('OBSERVER_RUNNER_DIAGNOSTIC');
        // A diagnostic containing an admitted token must never masquerade as child output.
        readChunk(file ?? '<runner>', type, data.message + '\n');
      }
    }
    if (leftovers.size !== 0) throw Error('OBSERVER_UNTERMINATED_LINE');
    emit('end', { children: constructed });
  } finally {
    lifecycle.unsubscribe(watchChild);
  }
}
