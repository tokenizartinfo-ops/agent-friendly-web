function hostCommand(exe,args,{limit=1048576,input}={}) {
  const remaining=jobStarted+65000-clock();
  assert(remaining>0);
  const out=spawnSync(exe,args,{cwd:host,env:hostEnv,encoding:null,input,maxBuffer:limit,timeout:Math.min(remaining,30000),stdio:['pipe','pipe','pipe']});
  assert(!out.error&&out.signal===null&&out.status===0&&out.stdout.length<=limit&&out.stderr.length<=65536);
  return out.stdout;
}
const git=(...args)=>exactDecoder.decode(hostCommand('/usr/bin/git',args)).trimEnd();

async function exists(path) {
  try { return await lstat(path); } catch(error){if(error.code==='ENOENT')return null;throw error;}
}
function admittedPath(root,path) {
  assert(safePath.test(path)&&path.split('/').every(x=>x!=='.'&&x!=='..'));
  const full=join(root,path);assert(full.startsWith(root+sep));return full;
}
async function pinned(path,p) {
  const old=await lstat(path);
  assert(old.isFile()&&await realpath(path)===path&&old.size===p.bytes);
  const data=await readFile(path), after=await lstat(path);
  assert(old.dev===after.dev&&old.ino===after.ino&&old.ctimeMs===after.ctimeMs&&data.length===p.bytes&&gitBlob(data)===p.blob&&sha256(data)===p.sha256);
  return data;
}

async function beforeLaunch(manifest,cwd,temp) {
  const archive=hostCommand('/usr/bin/git',['archive','--format=tar',base,'--',...baseArchive],{limit:archiveCap});
  hostCommand('/usr/bin/tar',['-xf','-','-C',cwd],{limit:65536,input:archive});
  const lock=await readFile(join(host,'package-lock.json'));
  assert.equal(gitBlob(lock),manifest.baselineRechecks['package-lock.json']);
  const dependencyLock=JSON.parse(lock).packages;
  assert.equal(dependencyLock['node_modules/@cloudflare/workers-oauth-provider'].version,'1.2.1');
  assert.equal(dependencyLock['node_modules/miniflare'].version,'5.20260831.0-alpha');
  assert.equal(dependencyLock['node_modules/workerd'].version,'1.20260831.1');
  assert.equal(dependencyLock['node_modules/esbuild'].version,'0.28.2');
  for(const [file,sha] of Object.entries(manifest.baselineRechecks)){
    const path=admittedPath(cwd,file);assert.equal(gitBlob(await readFile(path)),sha);
  }
  const seen=new Set();
  for(const [relative,pin] of Object.entries(manifest.objects)) {
    const bytes=await pinned(admittedPath(host,prefix+relative),pin);
    for(const target of pin.targets){
      assert(!seen.has(target));seen.add(target);
      const path=admittedPath(cwd,target);
      if(pin.before){
        const old=await readFile(path);
        assert(old.length===pin.before.bytes&&gitBlob(old)===pin.before.blob);
      }else assert.equal(await exists(path),null);
      await mkdir(dirname(path),{recursive:true,mode:0o700});
      await writeFile(path,bytes,{flag:pin.before?'w':'wx',mode:0o600});
    }
  }
  assert(await realpath(join(host,'node_modules'))===join(host,'node_modules'));
  await symlink(join(host,'node_modules'),join(cwd,'node_modules'),'dir');
  return {targetCount:seen.size};
}

  for(const [file,sha] of Object.entries(manifest.baselineRechecks))assert.equal(gitBlob(await readFile(admittedPath(dir,file))),sha);
}
