async function open(){
 const f=await initialEntryHarness('first-connection-publication',{coupledStages:true});
 try{await f.stub.configure('normal');assert.equal((await plain(f.stub.invoke(f.refs))).status,'material_recorded');await f.stub.restoreMethods();await f.mountFirstConnection();
  f.selectionSource=(await plain(f.stub.capsuleFacts())).material.ownerSelection;f.token=await f.enrollmentToken(f.selectionSource);return f;
 }catch(error){await close(f);throw error;}
}
async function close(f){if(!f)return;try{await f.stub.firstConnectionRestore();await f.stub.c2ScopeRestore();await f.stub.restoreMethods();}finally{await f.mf.dispose();await f.cleanup();}}
const discard=async r=>{const status=r.status;await r.body?.cancel();return status;};
const headers=(f,cookie=f.choice?.cookie)=>({'Cf-Access-Jwt-Assertion':f.token,...cookie?{Cookie:cookie}:{}});
async function choose(f){
 const r=await f.call('/connections/finite-prepare',{headers:headers(f)}),status=r.status,text=await r.text();assert.equal(status,200);
 const handle=/name="handle" value="([a-f0-9]{64})"/.exec(text)?.[1],projectRef=/name="projectRef" value="([a-f0-9]{64})"/.exec(text)?.[1],cookie=r.headers.get('set-cookie')?.split(';')[0];assert(handle&&projectRef&&cookie);return {handle,projectRef,cookie,text};
}
const post=(f,form,patch={})=>f.call('/connections/finite-prepare',{method:'POST',headers:{...headers(f,form.cookie),Origin:O,'Content-Type':'application/x-www-form-urlencoded',...patch.headers},body:patch.body??new URLSearchParams({handle:form.handle,projectRef:form.projectRef,decision:'approve'})});
async function count(f){const rows=await plain(f.stub.snapshot());return {upload:f.uploads.length-1,activation:f.activations.length-1,receipts:records(rows).filter(([k])=>k.endsWith(':receipt')).length};}
async function waitHeld(f){
 const end=performance.now()+30000;for(;;){const s=await plain(f.stub.firstConnectionState());if(s.fetchHeld)return s;if(performance.now()>=end)throw Error('SYN c2 native hold missing');await new Promise(r=>setTimeout(r,10));}
}
async function settled(f){const end=performance.now()+30000;for(;;){if((await plain(f.stub.firstConnectionAvailability())).available)return;if(performance.now()>=end)throw Error('SYN c2 native settlement missing');await new Promise(r=>setTimeout(r,10));}}
export async function createFiniteCoupledOwnerWorld(){
 const main=await open();let active=main,pendingForm=null,firstCookie=null,negative=null;
 const isolate=async()=>{await close(negative);negative=await open();active=negative;return negative;};
 const noFurther=async(f,before,response)=>{assert.notEqual(await discard(response),200);assert.deepEqual(await count(f),before);};
 return {
  async close(){await close(negative);await close(main);},
  counts(){return count(active);},
  async approve(stage){
   active=main;const f=main;
   if(stage==='connection'){
    const response=await approveFirstConnection(f);assert.equal(await discard(response),200);firstCookie=f.choice.cookie;
   }else{
    const form=pendingForm??await choose(f);pendingForm=null;
    if(stage==='finite')assert(form.text.includes('revocar únicamente la concesión OAuth creada aquí.'));
    assert.equal(await discard(await post(f,form)),200);f.choice=form;
   }
   const rows=await plain(f.stub.snapshot()),receipts=records(rows).filter(([k])=>k.endsWith(':receipt')).map(([,raw])=>JSON.parse(raw));
   assert.equal(receipts.length,['connection','notices','finite'].indexOf(stage)+1);
   const ordered=['connection','notices','finite'].slice(0,receipts.length).map(s=>receipts.find(r=>r.stage===s));assert(ordered.every(Boolean));
   assert.equal(new Set(ordered.map(r=>r.versionId)).size,ordered.length);assert.equal(new Set(ordered.map(r=>r.deploymentId)).size,ordered.length);
   for(let i=0;i<ordered.length;i++){
    const r=ordered[i],previous=i?ordered[i-1]:null;
    if(previous){assert.equal(r.baselineVersionId,previous.versionId);assert.equal(r.baselineDeploymentId,previous.deploymentId);}
    for(const effect of ['upload','activation']){
     const key='afw.finite-policy-effect.v1:'+identityHash({kind:'connection',preconnectionId:r.preconnectionId,sourceRevision:r.sourceRevision,stage:r.stage,effect});
     const send=JSON.parse(rows.find(([k])=>k===key+':send')[1]),ack=JSON.parse(rows.find(([k])=>k===key+':ack')[1]);
     assert.equal(ack.sendHash,hash('ConnectionSend',send));assert.equal(ack.resourceId,effect==='upload'?r.versionId:r.deploymentId);assert(ack.acknowledgedAt<send.deadline);
    }
    assert.equal(hash('ConnectionReceipt',r),r.receiptHash);
   }
   assert.equal(f.uploads.length-1,ordered.length);assert.equal(f.activations.length-1,ordered.length);
   // Mount the actual provider-returned version/configuration, with unchanged actual compiled M bytes.
   await f.adoptConnectionConfiguration({upstream:true});
  },
  async oauth(){
   const f=main,current=await plain(f.stub.firstConnectionContext());assert.equal(current.source.phase,'read_consent');f.source=current.source;
   const kv=await f.mf.getKVNamespace('OAUTH_KV','canary-first-connection');
   await kv.put('client:'+f.source.clientId,JSON.stringify({clientId:f.source.clientId,redirectUris:['https://client.example/cb'],clientName:'SYN c2 native client',grantTypes:['authorization_code'],responseTypes:['code'],tokenEndpointAuthMethod:'none',authMethodExplicit:true,registrationDate:Math.floor(Date.now()/1000)}));
   const verifier='v'.repeat(64),challenge=createHash('sha256').update(verifier).digest('base64url'),r=await f.call('/authorize?'+new URLSearchParams({client_id:f.source.clientId,redirect_uri:'https://client.example/cb',response_type:'code',resource:f.source.resource,scope:'afw:project:read',state:'SYN-c2',code_challenge:challenge,code_challenge_method:'S256'}),{redirect:'manual',headers:headers(f)});
   assert.equal(r.status,200);const text=await r.text(),handle=/name="handle" value="([^"]+)"/.exec(text)?.[1],cookie=r.headers.getSetCookie().map(v=>v.split(';')[0]).join('; ');assert(handle&&cookie);
   const approved=await f.call('/authorize',{method:'POST',redirect:'manual',headers:{...headers(f,firstCookie+'; '+cookie),Origin:O,'Content-Type':'application/x-www-form-urlencoded'},body:new URLSearchParams({handle,decision:'approve',scope:'afw:project:read'})});
   assert.equal(approved.status,302);assert(new URL(approved.headers.get('Location')).searchParams.get('code'));await approved.body?.cancel();
   const rows=await plain(f.stub.snapshot());assert(rows.some(([k])=>k.endsWith(':sdk_complete:observed')));
  },
  async invalidForm(key){
   const f=main,form=pendingForm??await choose(f);pendingForm=form;const before=await count(f),body=new URLSearchParams({handle:form.handle,projectRef:form.projectRef,decision:'approve'});body.set(key,'f'.repeat(64));await noFurther(f,before,await post(f,form,{body}));
  },
  async invalidSubject(){const f=main,before=await count(f),token=await f.firstConnectionToken(f.source,{subject:'SYN-c2-foreign'});await noFurther(f,before,await f.call('/connections/finite-prepare',{headers:{...headers(f),'Cf-Access-Jwt-Assertion':token}}));},
  async corruptThenDeny(kind){

   }finally{await f.stub.c2Restore(snapshot);f.providerRegistry.pointer=pointer;f.providerRegistry.versions=versions;if(grant)await f.db.prepare('UPDATE delegated_access_grants SET revoked_at=? WHERE id=?').bind(grant.revoked_at,grant.id).run();}
  },

  async monitor(){
