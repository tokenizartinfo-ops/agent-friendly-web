const fixture=name=>JSON.parse(readFileSync(new URL(name,import.meta.url),'utf8'));
const {focusedData:data}=fixture('root-ruling.json');
const {epoch,firstKey,secondKey}=data;
const response=(key,tuple)=>'q1.'+epoch+'.'+key+'.'+tuple;
const originalEpoch='1133557799bbddff', originalKey='11111111-1111-4111-8111-111111111111';

function context(create=model,{enabled=true, p={},register}={}) {
  const a={},b={};
  const qa=create({enabled,epoch,directProof:p,registerToolBudgetSeed:register});
  return {a,b,p,qa,add(first=true,second=true){if(first)qa.arrive(a,firstKey,p);if(second)qa.arrive(b,secondKey,p);},
    readA:()=>qa.sample(epoch,firstKey,p),readB:()=>qa.sample(epoch,secondKey,p)};
}

function primaryEntered(qa,request) {
  const options={};
  qa.phase(request,'B');qa.phase(request,'I');qa.enterPair(request,options,'initial');
  qa.aggregate(request,options,'facts1','enter');qa.aggregate(request,options,'facts1','return');
  qa.aggregate(request,options,'primary1','enter');
}

function overlapCase(id) {
  let seed;
  const c=id==='O08'?context(overflowTool,{register:f=>{seed=f;}}):
    id==='O10'?context(model,{enabled:false}):context();
  const {a,b,p,qa,readA,readB}=c;
  c.add(id!=='O04',id!=='O05');
  switch(id) {
    case 'O01': qa.overlap(a,b,p);assert.equal(readA(),'q0.O');assert.equal(readB(),'q0.O');break;
    case 'O02': primaryEntered(qa,a);primaryEntered(qa,b);qa.overlap(a,b,p);assert.equal(readA(),'q0.O');assert.equal(readB(),'q0.O');break;
    case 'O03': qa.overlap(a,b,{});assert.equal(readA(),response(firstKey,'A0000'));assert.equal(readB(),response(secondKey,'A0000'));break;
    case 'O04': qa.overlap(a,b,p);assert.equal(readA(),'q0.A');assert.equal(readB(),response(secondKey,'A0000'));break;
    case 'O05': qa.overlap(a,a,p);assert.equal(readA(),response(firstKey,'A0000'));break;
    case 'O06': qa.reaction(a,'F');qa.overlap(a,b,p);assert.equal(readA(),response(firstKey,'A000F'));assert.equal(readB(),response(secondKey,'A0000'));break;
    case 'O07': qa.overlap(a,b,p);qa.overlap(a,b,p);assert.equal(readA(),'q0.O');assert.equal(readB(),'q0.O');break;
    case 'O08': assert.equal(seed(a),true);qa.phase(a,'B');qa.overlap(a,b,p);assert.equal(readA(),'q0.X');assert.equal(readB(),response(secondKey,'A0000'));break;
    case 'O09': assert.equal(readA(),response(firstKey,'A0000'));qa.overlap(a,b,p);assert.equal(readB(),response(secondKey,'A0000'));assert.equal(readA(),'q0.O');break;
    case 'O10': qa.overlap(a,b,p);assert.equal(readA(),null);break;
    default: assert.fail('unexpected overlap fixture');
  }
}

function seedCase(id) {
  let seed;
  const c=context(id==='S06'?ordinary:overflowTool,{enabled:id!=='S04',register:cap=>{seed=cap;}});
  const {a,qa,readA}=c;c.add(true,false);
  switch(id) {
    case 'S01': assert.equal(seed(a),true);qa.phase(a,'B');assert.equal(readA(),'q0.X');break;
    case 'S02': assert.equal(seed(a),true);assert.equal(seed(a),false);qa.phase(a,'B');assert.equal(readA(),'q0.X');break;
    case 'S03': assert.equal(seed({}),false);assert.equal(seed(a),true);qa.phase(a,'B');assert.equal(readA(),'q0.X');break;
    case 'S04': assert.equal(seed,undefined);assert.equal(readA(),null);break;
    case 'S05': qa.reaction(a,'F');assert.equal(seed(a),false);assert.equal(readA(),response(firstKey,'A000F'));break;
    case 'S06': assert.equal(seed,undefined);qa.phase(a,'B');assert.equal(readA(),response(firstKey,'B0000'));break;
    default: assert.fail('unexpected seed fixture');
  }
}

async function mainBindingControl() {
  assert.equal(globalThis.__AFW_PENDING_GET_EPOCH__,undefined);
  globalThis.__AFW_PENDING_GET_EPOCH__=epoch;
  const link=await import('../../test/fixtures/pending-get/m-link.mjs');
  delete globalThis.__AFW_PENDING_GET_EPOCH__;
  const {pendingWinner:w,pendingBeginResponse:begin}=link;
  const {binding:{independentControls}}=fixture('gate-ruling.json');
  const labels=[];
  for(const [index,[id]] of independentControls.entries()){
    assert.equal(id,'W0'+String(index+1));
    const request=new Request('https://delegated-canary.agentfriendlyweb.dev/connections/finite-prepare');
    const lease={},check=()=>{},first=()=>{},response=new Response(null),error=new Error('SYN deadline');
    const cell=link.pendingMainEnter(request);link.pendingMainBind(request,lease);
    w.pendingM.mark(lease,'requestBytesEnter');w.pendingM.mark(lease,'requestBytesReturn');
    assert.equal(link.pendingWireCall(request,check,first),null);
    w.pendingM.captureUuid(lease,firstKey);w.pendingM.mark(lease,'stubEnter');w.pendingM.mark(lease,'stubReturn',response);
    if(id==='W03'){
      const foreignLease={};const foreignResponse=new Response(null);
      begin(foreignLease,foreignResponse,check,first);
      assert.equal(link.pendingWireCall(foreignResponse,check,first),null);
    }
    begin(lease,response,check,first);
    if(id==='W01')begin(lease,response,check,first);
    if(id==='W02'){
      const replacement=new Response(null);begin(lease,replacement,check,first);
      assert.equal(link.pendingWireCall(replacement,check,first),null);
    }
    assert.equal(link.pendingWireCall(response,()=>{},first),null);
    assert.equal(link.pendingWireCall(response,check,first),lease);
    assert.equal(link.pendingWireCall(response,check,first),null);

    labels.push(id);
  }
  return labels;
}

export async function runCalibrationModels() {
  const {vectors,shared}=fixture('QA-vectors.json');
  const labels=[];
  for(const [i,row] of vectors.entries()){
    assert.equal(row[0],'Q'+String(i+1).padStart(2,'0'));
    checkQa(row,{epoch:originalEpoch,uuid:originalKey},shared.m);
    labels.push(row[0]);
  }
  assert.equal(labels.length,35);
  for(const [i,[id]] of data.overlap.entries()){
    assert.equal(id,'O'+String(i+1).padStart(2,'0'));overlapCase(id);labels.push(id);
  }
  for(const [i,[id]] of data.seed.entries()){
    assert.equal(id,'S'+String(i+1).padStart(2,'0'));seedCase(id);labels.push(id);
  }
  assert.equal(labels.length,51);
  labels.push(...await mainBindingControl());
  assert.equal(labels.length,54);
  const {control}=fixture('lineage-ruling.json');assert.equal(control.id,'W04');
  await checkComposedLineage(control);labels.push(control.id);
  return {labels};
}
