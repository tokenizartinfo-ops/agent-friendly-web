function snapshotBucket(rows,main) {
 const unknown='UNKNOWN';
 if(!Array.isArray(rows)||rows.length<1||rows.length>512)return unknown;
 const currentProject=main.choice?.projectRef,currentRevision=main.source?.sourceRevision;
 if(typeof currentProject!=='string'||!/^[a-f0-9]{64}$/.test(currentProject)||
    typeof currentRevision!=='string'||!/^[a-f0-9]{40}$/.test(currentRevision))return unknown;
 const keys=new Set(),tickets=new Map(),uses=new Map();
 let project=null,ownerRows=0;
 for(const row of rows){
  if(!Array.isArray(row)||row.length!==2)return unknown;
  const [key,raw]=row;
  if(typeof key!=='string'||!key.length||key.length>1024||Buffer.byteLength(key)>1024||keys.has(key))return unknown;
  keys.add(key);
  const map=key.startsWith('owner-project-ref:');
  if(!map&&!key.startsWith('owner-ticket:'))continue;
  if(typeof raw!=='string'||raw.length>8192||Buffer.byteLength(raw)>8192)return unknown;
  if(map){
   if(project||key!=='owner-project-ref:'+currentProject)return unknown;
   project=decode('ProjectRef',raw);
   if(project.projectRef!==currentProject||project.sourceRevision!==currentRevision)return unknown;
   continue;
  }
  if(++ownerRows>16)return unknown;
  const match=/^owner-ticket:([a-f0-9]{64})(:use)?$/.exec(key);
  if(!match)return unknown;
  const item=decode(match[2]?'TicketUse':'Ticket',raw);
  if(item.ticketHash!==match[1])return unknown;
  (match[2]?uses:tickets).set(item.ticketHash,item);
 }
 if(!project)return unknown;
 let connection=null,notices=null,finite=0;
 for(const t of tickets.values()){
  if(t.projectRef!==project.projectRef||t.sourceRevision!==project.sourceRevision||t.preconnectionId!==project.preconnectionId||
     t.subjectHash!==project.subjectHash||t.createdAt<project.issuedAt||t.expiresAt>project.expiresAt)return unknown;
  if(t.action==='connection'){if(connection||t.connectionRef!==null)return unknown;connection=t;}
  if(t.action==='notices'){if(notices||!t.connectionRef)return unknown;notices=t;}
 }
 if(!connection||!notices||!uses.has(connection.ticketHash)||!uses.has(notices.ticketHash))return unknown;
 for(const [hash,u] of uses){
  const t=tickets.get(hash);
  if(!t||u.consumedAt<t.createdAt||u.consumedAt>=t.expiresAt)return unknown;
 }
 for(const t of tickets.values()){
  if(t.action==='finite'){if(t.connectionRef!==notices.connectionRef)return unknown;finite++;}
 }
 return finite===0?'0':finite===1?'1':'2+';
}

export function createFiniteSnapshotObserver(main,getActive,emit=message=>console.info(message)) {
 const get=createGetWinnerHost(main,()=>main.getWinnerDirect?.()===true,DispatchResponse,pendingBoundary?.park);let getToken=null;
 let state='idle',pending=null;
 function enter(id) {
  if(state!=='idle'||!['C2T1_MONITOR','C2T1_SCOPE_MISSING'].includes(id)){
   state='poisoned';pending=null;return null;
  }
  pending={id};state='active';if(id==='C2T1_SCOPE_MISSING')getToken=get.begin(pending);return pending;
 }
 function finish(label) {
  if(state==='active'&&label===pending){state='ready';if(label.id==='C2T1_SCOPE_MISSING')get.finish(label);}
  else {state='poisoned';pending=null;get.poison();}
 }
 function counts() {
  const active=getActive();
  if(state==='active'){state='poisoned';pending=null;get.poison();}
  if(state!=='ready')return count(active);
  const label=pending;pending=null;state='idle';
  if(active!==main)return count(active);
  return count(active,rows=>{
   let value='UNKNOWN';
   try{value=snapshotBucket(rows,main);}catch{}
   try{
    if(label.id==='C2T1_SCOPE_MISSING') {
     const winner=get.take(label,active)??{w:'U',s:0,h:0};
     emit('C2FT4 '+JSON.stringify({id:label.id,value,...winner}));
    } else emit('C2FT3 '+JSON.stringify({id:label.id,value}));
   }catch{}
  });
 }
