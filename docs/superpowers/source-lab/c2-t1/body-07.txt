import test,{after} from 'node:test';
import assert from 'node:assert/strict';
import {createFiniteCoupledOwnerWorld} from './fixtures/finite-coupled-native.mjs';
let w;
after(async()=>{await w?.close();});
const fixed=(id,phase,work)=>test(id,{timeout:120000},async()=>{
 assert(w||id==='C2T1_CONNECTION');
 let actionThrew=false,counterThrew=false,actionError,counterError,counts;
 try{await work();}catch(error){actionThrew=true;actionError=error;}
 try{counts=w?await w.counts():null;}catch(error){counterThrew=true;counterError=error;}
 // Missing counter facts cannot produce a Source frame or a qualifying receipt.
 if(!counterThrew&&counts!==null&&counts!==undefined){
  const passed=!actionThrew;
  console.info('C2SOURCE '+JSON.stringify({id,status:passed?'pass':'fail',phase,counters:counts,code:passed?'OK':'TEST_FAILED'}));
 }
 // Preserve the exact original thrown value, including null or undefined.
 if(actionThrew)throw actionError;
 if(counterThrew)throw counterError;
});
fixed('C2T1_CONNECTION','connection',async()=>{
 w=await createFiniteCoupledOwnerWorld();
 await w.approve('connection');
 assert.deepEqual(await w.counts(),{upload:1,activation:1,receipts:1});
});
