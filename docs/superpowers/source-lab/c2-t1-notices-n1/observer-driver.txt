import test,{after as nativeAfter} from 'node:test';
import assert from 'node:assert/strict';
import {createFiniteCoupledOwnerWorld} from './fixtures/finite-coupled-native.mjs';
import {ready,flow,blocked,after,captureFault} from './n1-helper.mjs';
let w,prerequisiteFailed=false;
nativeAfter(async()=>{after('enter');try{await w?.close();after('return');}catch(error){after('throw',error);throw error;}});
const run=(id,work)=>test(id,{timeout:120000},async()=>{
 if(prerequisiteFailed){blocked(id);const error=Error('N1 prerequisite blocked');error.code='PREREQUISITE_BLOCKED';throw error;}
 let actionThrew=false,countThrew=false,actionError,countError;
 flow(id,'action','enter');
 try{assert(ready());if(id==='N1_NOTICES')await w.n1Before();await work();flow(id,'action','return');}
 catch(error){actionThrew=true;actionError=error;flow(id,'action','throw',error);}
 if(id==='N1_NOTICES'){try{await w.n1After();}catch{captureFault();}}
 flow(id,'count','enter');
 try{const counters=w?await w.counts():null;flow(id,'count','return',null,counters);}
 catch(error){countThrew=true;countError=error;flow(id,'count','throw',error);}
 if(actionThrew||countThrew)prerequisiteFailed=true;
 if(actionThrew)throw actionError;if(countThrew)throw countError;
});
run('N1_CONNECTION',async()=>{w=await createFiniteCoupledOwnerWorld();await w.approve('connection');assert.deepEqual(await w.counts(),{upload:1,activation:1,receipts:1});});
run('N1_OAUTH',async()=>{await w.oauth();});
run('N1_PHASE',async()=>{await w.corruptThenDeny('phase');});
run('N1_NOTICES',async()=>{await w.approve('notices');assert.deepEqual(await w.counts(),{upload:2,activation:2,receipts:2});});
